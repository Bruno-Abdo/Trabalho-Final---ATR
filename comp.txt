"""
Cliente MQTT Publisher para a Simulação da Mina.

Responsável por enviar os dados dos sensores do caminhão para o broker MQTT,
permitindo monitoramento externo (Etapa 2).

Requisitos atendidos:
    - Cliente MQTT Publisher 
    - Publicação de variáveis da Tabela 1 
"""

from __future__ import annotations

import json
import time
from typing import Optional

from PyQt6.QtCore import QObject, pyqtSignal

# Tentativa de importação segura para não quebrar a IDE se faltar a lib
try:
    import paho.mqtt.client as mqtt
    MQTT_AVAILABLE = True
except ImportError:
    MQTT_AVAILABLE = False
    print("[publisher] AVISO: paho-mqtt não instalado. Funcionalidade de rede desabilitada.")

from model.truck_state import TruckState


class TruckPublisher(QObject):
    """
    Cliente MQTT encapsulado em QObject.
    
    Signals:
        connected: Conexão estabelecida com sucesso.
        disconnected: Conexão perdida ou encerrada.
        published(topic, payload): Mensagem enviada com sucesso (útil para logs).
    """
    
    # Sinais para integração com a UI
    connected = pyqtSignal()
    disconnected = pyqtSignal()
    published = pyqtSignal(str, str)  # topic, payload
    
    def __init__(self, broker_host: str = "localhost", broker_port: int = 1883):
        super().__init__()
        
        self._broker_host = broker_host
        self._broker_port = broker_port
        self._connected = False
        self._client: Optional[mqtt.Client] = None
        
        if MQTT_AVAILABLE:
            # Client ID único para evitar desconexões por conflito
            client_id = f"sim_truck_{int(time.time())}_{id(self)}"
            self._client = mqtt.Client(client_id=client_id, protocol=mqtt.MQTTv311)
            
            # Configura callbacks do Paho MQTT
            self._client.on_connect = self._on_connect
            self._client.on_disconnect = self._on_disconnect
    
    # =======================================================================
    # CONTROLE DE CONEXÃO
    # =======================================================================
    
    def connect_to_broker(self) -> None:
        """Inicia a conexão assíncrona com o broker."""
        if not MQTT_AVAILABLE or self._client is None:
            return
            
        try:
            print(f"[Publisher] Conectando a {self._broker_host}:{self._broker_port}...")
            self._client.connect(self._broker_host, self._broker_port, keepalive=60)
            self._client.loop_start()  # Thread de rede independente
        except Exception as e:
            print(f"[Publisher] Erro ao tentar conectar: {e}")
            self._connected = False
            self.disconnected.emit()

    def disconnect_from_broker(self) -> None:
        """Encerra a conexão e o loop de rede."""
        if self._client is not None:
            self._client.loop_stop()
            self._client.disconnect()
            # O callback on_disconnect será chamado e emitirá o sinal
            
    def is_connected(self) -> bool:
        return self._connected

    # =======================================================================
    # CALLBACKS MQTT
    # =======================================================================

    def _on_connect(self, client, userdata, flags, rc):
        if rc == 0:
            self._connected = True
            self.connected.emit()
            print("[Publisher] Conectado.")
        else:
            self._connected = False
            print(f"[Publisher] Falha na conexão. Código: {rc}")

    def _on_disconnect(self, client, userdata, rc):
        self._connected = False
        self.disconnected.emit()
        if rc != 0:
            print(f"[Publisher] Desconexão inesperada. Código: {rc}")
        else:
            print("[Publisher] Desconectado.")

    # =======================================================================
    # PUBLICAÇÃO
    # =======================================================================

    def publish_state(self, truck_id: int, state: TruckState) -> None:
        """
        Publica o estado atual do caminhão.
        
        Tópico: atr/mine/truck/<id>/sensors
        Payload: JSON (com valores inteiros conforme Tabela 1)
        """
        if not self._connected or self._client is None:
            return

        topic = f"atr/mine/truck/{truck_id}/sensors"
        
        # quantize=True força conversão para int/bool conforme Tabela 1
        data = state.get_sensor_dict()
        
        try:
            payload = json.dumps(data)
            info = self._client.publish(topic, payload, qos=0)
            
            if info.rc == mqtt.MQTT_ERR_SUCCESS:
                self.published.emit(topic, payload)
                
        except Exception as e:
            print(f"[Publisher] Erro de publicação: {e}")
